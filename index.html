import sounddevice as sd
import numpy as np

def capture_audio_stream(duration=4, sample_rate=16000):
    audio = sd.rec(int(duration * sample_rate), samplerate=sample_rate, channels=1)
    sd.wait()
    return np.squeeze(audio)
